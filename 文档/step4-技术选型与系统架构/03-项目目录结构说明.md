## 二、后端目录（backend/）

```
backend/
├── pom.xml                                   # Maven 依赖（含 springdoc-openapi）
└── src/main/
    ├── java/com/example/campusmaster/
    │   │
    │   ├── CampusMasterApplication.java      # 启动类（@SpringBootApplication + @MapperScan + @EnableTransactionManagement）
    │   │
    │   ├── annotation/
    │   │   └── RequiresRoles.java            # 自定义注解：方法级角色校验
    │   │
    │   ├── common/                            # 通用基础类
    │   │   ├── Result.java                   # 统一响应 {code, message, data}
    │   │   ├── BusinessException.java        # 业务异常（含 conflict/unauthorized/forbidden 等工厂方法）
    │   │   ├── GlobalExceptionHandler.java   # @RestControllerAdvice 全局异常处理
    │   │   └── JwtUtil.java                   # JWT 生成/解析（jjwt 0.12.5）
    │   │
    │   ├── config/                            # 配置类
    │   │   ├── SecurityConfig.java           # Spring Security（CORS + 禁 CSRF + 无状态 + 放行 Swagger）
    │   │   ├── MyBatisPlusConfig.java         # MyBatis-Plus（分页插件、乐观锁插件）
    │   │   ├── WebMvcConfig.java             # CORS + 拦截器注册
    │   │   ├── WebSocketConfig.java           # WebSocket 端点注册
    │   │   └── OpenApiConfig.java             # 【待加】Swagger 配置（Step 3 产出）
    │   │
    │   ├── controller/                        # REST API 控制器
    │   │   ├── UserController.java           # /api/auth/** （登录、注册、用户信息、改密码）
    │   │   ├── TaskController.java            # /api/tasks/** （CRUD + 状态机 8 个动作）
    │   │   ├── AccountController.java         # /api/account/** （账户查询、流水、充值）
    │   │   ├── NotificationController.java    # /api/notifications/** （列表、未读数、标记已读）
    │   │   ├── ReviewController.java          # /api/reviews/** （评价 CRUD）
    │   │   └── AdminController.java           # /api/admin/** （统计、任务、争议、用户）
    │   │
    │   ├── entity/                            # 实体类（与 schema.sql 一一对应）
    │   │   ├── User.java                      # 用户（含 @TableLogic、@Version）
    │   │   ├── Task.java                      # 任务（含 audit_status、version）
    │   │   ├── Account.java                   # 账户（balance/frozenBalance/version）
    │   │   ├── Transaction.java               # 资金流水
    │   │   ├── Review.java                    # 评价
    │   │   ├── Notification.java              # 通知
    │   │   ├── Dispute.java                   # 争议
    │   │   └── TaskLog.java                   # 任务日志
    │   │
    │   ├── mapper/                            # MyBatis-Plus Mapper
    │   │   ├── UserMapper.java                # 含 selectByPhone 自定义
    │   │   ├── TaskMapper.java
    │   │   ├── AccountMapper.java             # 含 selectByUserId 自定义
    │   │   ├── TransactionMapper.java
    │   │   ├── ReviewMapper.java              # 含 selectByTaskIdAndReviewerId
    │   │   ├── NotificationMapper.java
    │   │   ├── DisputeMapper.java             # 含 selectByTaskId
    │   │   └── TaskLogMapper.java
    │   │
    │   ├── service/                           # 业务接口
    │   │   ├── UserService.java
    │   │   ├── TaskService.java
    │   │   ├── AccountService.java
    │   │   ├── NotificationService.java
    │   │   ├── ReviewService.java
    │   │   └── AdminService.java
    │   │
    │   ├── service/impl/                      # 业务实现
    │   │   ├── UserServiceImpl.java           # 注册/登录/密码/信用分
    │   │   ├── TaskServiceImpl.java           # 状态机 8 个动作 + TaskLog 记录
    │   │   ├── AccountServiceImpl.java        # 资金 4 个动作 + Transaction 记录
    │   │   ├── NotificationServiceImpl.java   # 落库 + WebSocket 推送
    │   │   ├── ReviewServiceImpl.java         # 评价 + 信用分联动
    │   │   └── AdminServiceImpl.java          # 统计/审核/争议裁决/冻结
    │   │
    │   ├── interceptor/                       # Spring MVC 拦截器
    │   │   ├── AuthInterceptor.java           # 解析 token, 注入 userId/role
    │   │   └── RoleInterceptor.java           # 检查 @RequiresRoles 注解
    │   │
    │   └── websocket/
    │       └── WebSocketServer.java            # /ws/{userId} 端点 + Session 池
    │
    └── resources/
        ├── application.yml                    # 主配置（端口/数据库/日志）
        ├── application-local.example.yml      # 本地开发模板
        ├── schema.sql                         # 8 张表 DDL
        ├── data.sql                           # 5 个测试用户 + 5 个账户
        └── mybatis-config.xml                 # MyBatis 基础配置
```

## 四、根目录配置文件
